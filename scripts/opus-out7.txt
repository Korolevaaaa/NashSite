```jsx
// ProjectorBeam.jsx
import { forwardRef, useImperativeHandle, useMemo, useRef } from 'react';

export const ProjectorBeam = forwardRef(function ProjectorBeam(
  { x0, y0, x1, y1, packets = 5 },
  ref
) {
  const lineRef = useRef(null);
  const ringRef = useRef(null);
  const packetRefs = useRef([]);

  const geom = useMemo(() => {
    const dx = x1 - x0;
    const dy = y1 - y0;
    const len = Math.hypot(dx, dy) || 1;
    const ux = dx / len;
    const uy = dy / len;
    const nx = -uy;
    const ny = ux;

    // геометрию и ширины не меняем — оставляем как в текущем коде
    const w0 = 3;
    const w1 = Math.max(26, Math.min(78, len * 0.17));

    return {
      x0,
      y0,
      x1,
      y1,
      len,
      w0,
      w1,
      quad: [
        [x0 + (nx * w0) / 2, y0 + (ny * w0) / 2],
        [x1 + (nx * w1) / 2, y1 + (ny * w1) / 2],
        [x1 - (nx * w1) / 2, y1 - (ny * w1) / 2],
        [x0 - (nx * w0) / 2, y0 - (ny * w0) / 2],
      ]
        .map((p) => p.join(','))
        .join(' '),
    };
  }, [x0, y0, x1, y1]);

  useImperativeHandle(ref, () => ({
    frame(t, progress = 1) {
      const p = Math.max(0, Math.min(1, progress));
      const flicker = 0.93 + 0.07 * Math.sin(t * 21.3);

      if (lineRef.current) {
        lineRef.current.style.opacity = ((0.24 + 0.46 * p) * flicker).toFixed(3);
        lineRef.current.style.strokeDashoffset = `${(-t * 54).toFixed(1)}px`;
      }

      if (ringRef.current) {
        const pulse = 1 + 0.06 * Math.sin(t * 2.4);
        ringRef.current.style.opacity = (0.16 + 0.84 * p).toFixed(3);
        ringRef.current.setAttribute(
          'transform',
          `translate(${geom.x1} ${geom.y1}) scale(${pulse.toFixed(3)})`
        );
      }

      packetRefs.current.forEach((node, i) => {
        if (!node) return;

        const speed = 0.14 + (i % 3) * 0.022;
        const offset = i / packetRefs.current.length;

        let k = (t * speed + offset) % 1;
        if (p < 0.995) k = Math.min(k, p);

        const x = geom.x0 + (geom.x1 - geom.x0) * k;
        const y = geom.y0 + (geom.y1 - geom.y0) * k;

        const twinkle = 0.45 + 0.55 * Math.sin(t * 18 + i * 2.39);
        const fade = k < 0.08 ? k / 0.08 : k > 0.9 ? (1 - k) / 0.1 : 1;

        node.setAttribute('transform', `translate(${x.toFixed(2)} ${y.toFixed(2)})`);
        node.style.opacity = Math.max(
          0,
          (0.18 + 0.72 * twinkle) * fade * p * flicker
        ).toFixed(3);
      });
    },
  }));

  return (
    <g className="pb">
      <polygon className="pb-base" points={geom.quad} />
      <line
        ref={lineRef}
        className="pb-line"
        x1={x0}
        y1={y0}
        x2={x1}
        y2={y1}
      />

      <g ref={ringRef} className="pb-ring" transform={`translate(${x1} ${y1})`}>
        <circle className="pb-ring-outer" r={Math.max(10, geom.w1 * 0.34)} />
        <circle className="pb-ring-core" r={Math.max(4, geom.w1 * 0.14)} />
      </g>

      <g className="pb-packets">
        {Array.from({ length: packets }).map((_, i) => (
          <g
            key={i}
            className="pb-packet"
            ref={(el) => (packetRefs.current[i] = el)}
          >
            <circle className="pb-packet-halo" r="4.4" />
            <circle className="pb-packet-core" r="1.8" />
          </g>
        ))}
      </g>
    </g>
  );
});
```

const DRAFT_KEY = 'storefront.draft.v1';
const PUBLISHED_KEY = 'storefront.published.v1';
const isBrowser = typeof window !== 'undefined';

function seedStore() {
  const categories = [];

  TEMPLATES.forEach((item) => {
    let cat = categories.find((c) => c.name === item.cat);

    if (!cat) {
      cat = { name: item.cat, subs: [] };
      categories.push(cat);
    }

    if (item.sub && !cat.subs.includes(item.sub)) {
      cat.subs.push(item.sub);
    }
  });

  return {
    categories,
    items: TEMPLATES,
    meta: {
      updatedAt: null,
      published: false,
    },
  };
}

function loadDraft() {
  if (!isBrowser) return seedStore();

  try {
    const raw = localStorage.getItem(DRAFT_KEY);
    return raw ? JSON.parse(raw) : seedStore();
  } catch {
    return seedStore();
  }
}

function loadPublished() {
  if (!isBrowser) return seedStore();

  try {
    const raw = localStorage.getItem(PUBLISHED_KEY);
    return raw ? JSON.parse(raw) : seedStore();
  } catch {
    return seedStore();
  }
}

function downloadJSON(data, filename = 'storefront.draft.json') {
  const blob = new Blob([JSON.stringify(data, null, 2)], {
    type: 'application/json',
  });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');

  a.href = url;
  a.download = filename;
  a.click();

  URL.revokeObjectURL(url);
}

function useHashRoute() {
  const [hash, setHash] = useState(isBrowser ? window.location.hash : '');

  useEffect(() => {
    const onChange = () => setHash(window.location.hash);
    window.addEventListener('hashchange', onChange);

    return () => window.removeEventListener('hashchange', onChange);
  }, []);

  return hash.replace(/^#/, '');
}

export function StorefrontRoot() {
  const route = useHashRoute();

  if (route === '/admin') {
    return <StorefrontAdmin />;
  }

  return <StorefrontCatalog />;
}

export function StorefrontCatalog() {
  const [store] = useState(loadPublished);
  const [cat, setCat] = useState('all');
  const [sub, setSub] = useState('all');
  const [q, setQ] = useState('');

  const categories = useMemo(() => {
    return store.categories.filter((c) =>
      store.items.some((i) => i.status === 'published' && i.cat === c.name)
    );
  }, [store.categories, store.items]);

  const subs = useMemo(() => {
    if (cat === 'all') return [];

    const current = store.categories.find((c) => c.name === cat);
    if (!current) return [];

    return current.subs.filter((s) =>
      store.items.some(
        (i) => i.status === 'published' && i.cat === cat && i.sub === s
      )
    );
  }, [cat, store.categories, store.items]);

  const items = useMemo(() => {
    const query = q.trim().toLowerCase();

    return store.items
      .filter((i) => i.status === 'published')
      .filter((i) => cat === 'all' || i.cat === cat)
      .filter((i) => sub === 'all' || i.sub === sub)
      .filter((i) => {
        if (!query) return true;

        return [i.name, i.cat, i.sub, i.kind]
          .filter(Boolean)
          .join(' ')
          .toLowerCase()
          .includes(query);
      });
  }, [store.items, cat, sub, q]);

  return (
    <section className="sf" id="storefront">
      <div className="sf-filters">
        <input
          className="sf-search"
          type="search"
          value={q}
          onChange={(e) => {
            setQ(e.target.value);
            setSub('all');
          }}
          placeholder="Поиск по витрине"
          aria-label="Поиск по витрине"
        />

        <div className="sf-chips">
          <button
            type="button"
            className={cat === 'all' ? 'chip chip--active' : 'chip'}
            onClick={() => {
              setCat('all');
              setSub('all');
            }}
          >
            Все
          </button>

          {categories.map((c) => (
            <button
              key={c.name}
              type="button"
              className={cat === c.name ? 'chip chip--active' : 'chip'}
              onClick={() => {
                setCat(c.name);
                setSub('all');
              }}
            >
              {c.name}
            </button>
          ))}
        </div>

        {cat !== 'all' && subs.length > 0 && (
          <div className="sf-chips sf-chips--sub">
            <button
              type="button"
              className={sub === 'all' ? 'chip chip--active' : 'chip'}
              onClick={() => setSub('all')}
            >
              Все
            </button>

            {subs.map((s) => (
              <button
                key={s}
                type="button"
                className={sub === s ? 'chip chip--active' : 'chip'}
                onClick={() => setSub(s)}
              >
                {s}
              </button>
            ))}
          </div>
        )}
      </div>

      <div className="sf-grid">
        {items.map((item) => (
          <TemplateCard key={item.id} item={item} />
        ))}
      </div>

      {items.length === 0 && <p className="sf-empty">Ничего не найдено.</p>}
    </section>
  );
}
